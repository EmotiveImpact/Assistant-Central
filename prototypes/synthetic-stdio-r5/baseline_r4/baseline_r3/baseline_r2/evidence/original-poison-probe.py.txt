from independent_peer.fixture import PeerProcess, UncertainOutcome
from independent_peer.protocol import Identity,CAPABILITY,digest,token,scope,seal,open_packet
with tempfile.TemporaryDirectory() as directory:
 p=PeerProcess('alice',directory)
 try:
  pub=p.owner('describe')['descriptor']; bad=Identity('bob')
  task={'id':token(),'context':'0'*64,'capability':CAPABILITY}
  p.owner('configure',task=task,peer=bad.descriptor); incoming=p.owner('allow_receive')['grant']

  result={'id':token(),'scope':scope(task,'bob','alice','result'),'grant':incoming,'request':{'id':body['id'],'digest':digest(body),'processed':proof},'payload':'fiction'}
  try: p.call('owner',{'op':'inspect'},timeout=0)
  except UncertainOutcome: pass
  print('wrapper_poisoned',p.broken,'child_still_alive',p.process.poll() is None)
  # Real remote ingress survives client-wrapper poison; explicit low-level test only.
  response=p._io(p.peer_r,p.peer_w,{'op':'deliver','value':seal(bad,pub,bad.sign('message',result))},3)
  print('peer_delivery_after_poison',response['ok'])
  print('database_exists_before_close',os.path.exists(p.database))
 finally: p.close()
 print('child_reaped_after_close',p.process.poll() is not None)
